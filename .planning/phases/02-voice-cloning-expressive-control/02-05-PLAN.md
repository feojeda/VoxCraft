   ```typescript
   export interface PronunciationEntry {
     id: string;
     word: string;
     replacement: string;
   }

   export async function createPronunciationEntry(word: string, replacement: string): Promise<PronunciationEntry> {
     const res = await fetch("/api/pronunciation", {
       method: "POST",
       headers: { "Content-Type": "application/json" },
       body: JSON.stringify({ word, replacement }),
     });
     if (!res.ok) throw new Error(await res.text());
     return res.json();
   }

   export async function listPronunciationEntries(): Promise<PronunciationEntry[]> {
     const res = await fetch("/api/pronunciation");
     if (!res.ok) throw new Error("Failed to fetch pronunciation entries");
     return res.json();
   }

   export async function deletePronunciationEntry(entryId: string): Promise<void> {
     const res = await fetch(`/api/pronunciation/${entryId}`, { method: "DELETE" });
     if (!res.ok) throw new Error("Failed to delete entry");
   }
   ```
